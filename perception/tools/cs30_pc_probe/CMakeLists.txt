cmake_minimum_required(VERSION 3.22)
if(POLICY CMP0141)
  cmake_policy(SET CMP0141 NEW)
endif()
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
project(cs30_pc_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPOSITORY "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LINGTOUCH_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LINGTOUCH_WITH_LIBUSB OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPOSITORY}/vision/core" repository-core)
if(NOT CS30_USB_LIBRARY)
  message(FATAL_ERROR "Set CS30_USB_LIBRARY to a libusb import library")
endif()
add_executable(cs30_pc_probe main.cpp "${REPOSITORY}/vision/native/cs30_libusb/src/cs30_libusb_source.cpp")
target_include_directories(cs30_pc_probe PRIVATE "${REPOSITORY}/vision/native/cs30_libusb/include"
  "${REPOSITORY}/native/third_party/libusb/libusb")
target_compile_definitions(cs30_pc_probe PRIVATE NOMINMAX)
target_link_libraries(cs30_pc_probe PRIVATE lingtouch_engine "${CS30_USB_LIBRARY}")
lingtouch_target_defaults(cs30_pc_probe)
